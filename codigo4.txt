import { useState } from "react";
import { Link, useNavigate, useSearchParams } from "react-router-dom";
import { restablecerContrasena } from "../services/api";
import { Alerta, Boton, Campo } from "../components/ui";

function RestablecerContrasena() {
    const [parametros] = useSearchParams();
    const token = parametros.get("token");
    const navigate = useNavigate();

    const [contrasena, setContrasena] = useState("");
    const [confirmacion, setConfirmacion] = useState("");
    const [verContrasena, setVerContrasena] = useState(false);
    const [error, setError] = useState(null);
    const [cargando, setCargando] = useState(false);

    async function manejarEnvio(evento) {
        evento.preventDefault();
        setError(null);

        if (contrasena.length < 8) {
            setError("La contraseÃ±a debe tener al menos 8 caracteres.");
            return;
        }
        if (contrasena !== confirmacion) {
            setError("Las contraseÃ±as no coinciden.");
            return;
        }

        setCargando(true);
        try {
            await restablecerContrasena(token, contrasena);
            void navigate("/login", { state: { restablecida: true }, replace: true });
        } catch (err) {
            setError(err.message);
            setCargando(false);
        }
    }

    return (
        <div className="flex justify-center px-4 py-12">
            <div className="w-full max-w-sm">
                <div className="text-center mb-6">
                    <h2 className="text-2xl font-semibold text-gray-900">Elige una contraseÃ±a nueva</h2>
                </div>

                <div className="bg-white rounded-2xl border border-gray-200 shadow-sm p-6">
                    {!token ? (
                        <div className="text-center">
                            <Alerta tipo="error">
                                El enlace estÃ¡ incompleto. Abre el enlace completo del correo o pide uno nuevo.
                            </Alerta>
                        </div>
                    ) : (
                        <form onSubmit={manejarEnvio} className="flex flex-col gap-4">
                            <Campo
                                etiqueta="ContraseÃ±a nueva"
                                type={verContrasena ? "text" : "password"}

                                onChange={(e) => setContrasena(e.target.value)}
                                required
                                minLength={8}
                                autoComplete="new-password"
                                autoFocus
                                ayuda="MÃ­nimo 8 caracteres."
                            />
                            <Campo
                                etiqueta="Repite la contraseÃ±a"
                                type={verContrasena ? "text" : "password"}
                                value={confirmacion}
                                onChange={(e) => setConfirmacion(e.target.value)}

                                autoComplete="new-password"
                            />

                                <input

                                    checked={verContrasena}

                                />

import { useEffect, useState } from "react";
import { reenviarVerificacion } from "../services/api";
import { Alerta, Boton, Campo } from "./ui";

const ESPERA_SEGUNDOS = 60;

function ReenviarVerificacion({ correo: correoFijo }) {
    const [correoEscrito, setCorreoEscrito] = useState("");
    const [enviando, setEnviando] = useState(false);
    const [espera, setEspera] = useState(0);
    const [mensaje, setMensaje] = useState(null);
    const [error, setError] = useState(null);

    const correo = correoFijo ?? correoEscrito;

    useEffect(() => {
        if (espera <= 0) return;
        const temporizador = setTimeout(() => setEspera((segundos) => segundos - 1), 1000);
        return () => clearTimeout(temporizador);
    }, [espera]);

    async function reenviar(evento) {
        evento.preventDefault();
        setError(null);
        setMensaje(null);
        setEnviando(true);

        try {
            const respuesta = await reenviarVerificacion(correo);
            setMensaje(respuesta.mensaje);
            setEspera(ESPERA_SEGUNDOS);
        } catch (err) {
            setError(err.message);
        } finally {
            setEnviando(false);
        }
    }

    return (
        <form onSubmit={reenviar} className="mt-4 flex flex-col gap-3 text-left">
            {correoFijo === undefined && (
                <Campo
                    etiqueta="Correo electrÃ³nico"
                    type="email"
                    value={correoEscrito}
                    onChange={(e) => setCorreoEscrito(e.target.value)}
                    required
                    autoComplete="email"
                />
            )}

            <Boton
                type="submit"
                variante="secundario"
                cargando={enviando}
                disabled={espera > 0 || !correo.trim()}
            >
                {espera > 0 ? `Reenviar en ${espera} s` : "Reenviar correo de confirmaciÃ³n"}
            </Boton>

            <Alerta tipo="exito">{mensaje}</Alerta>
            <Alerta tipo="error">{error}</Alerta>
        </form>
    );
}

export default ReenviarVerificacion;
